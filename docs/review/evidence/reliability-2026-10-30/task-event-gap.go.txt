package server

import (
	"context"
	"errors"
	"github.com/cuihairu/croupier/internal/tasks"
	sdkv1 "github.com/cuihairu/croupier/pkg/pb/croupier/sdk/v1"
	"github.com/stretchr/testify/require"
	"google.golang.org/protobuf/proto"
	"testing"
)

type reviewEventGapStore struct {
	terminal bool
	attempts int
}

func (s *reviewEventGapStore) UpdateRunIfStatusNotIn(_ context.Context, _ string, _ []string, _ map[string]interface{}) (bool, error) {
	if s.terminal {
		return false, nil
	}
	s.terminal = true
	return true, nil
}
func (s *reviewEventGapStore) AppendEvent(_ context.Context, _ string, _ tasks.EventType, _ int32, _ string, _ []byte) error {
	s.attempts++
	return errors.New("injected append failure")
}
func TestReview20261030_TerminalRetrySkipsFailedAppend(t *testing.T) {
	s := newTestControlService()
	store := &reviewEventGapStore{}
	s.SetTaskStore(store)
	b, err := proto.Marshal(&sdkv1.TaskEvent{TaskId: "review-task", Type: "completed", Payload: []byte(`{"ok":true}`)})
	require.NoError(t, err)
	_, err = s.handleTaskEvent(context.Background(), b)
	require.Error(t, err)
	require.True(t, store.terminal)
	_, err = s.handleTaskEvent(context.Background(), b)
	require.NoError(t, err)
	require.Equal(t, 1, store.attempts)
	t.Log("CONFIRMED first append failed; retry ACKed without another append")
}
