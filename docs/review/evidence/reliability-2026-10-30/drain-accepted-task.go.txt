package agent

import (
	"context"
	"github.com/cuihairu/croupier/agents/sidecar"
	sdkv1 "github.com/cuihairu/croupier/pkg/pb/croupier/sdk/v1"
	"github.com/stretchr/testify/require"
	"google.golang.org/protobuf/proto"
	"testing"
	"time"
)

func TestReview20261030_DrainLosesAcceptedQueuedTask(t *testing.T) {
	h, store := newDrainTestHandler(t)
	sess := newEchoSession(t, "review-session", "review-service", "fn.review", nil, nil)
	registerDrainSession(t, h, store, sess, "fn.review")
	entered, release := make(chan struct{}), make(chan struct{})
	finished := make(chan error, 1)
	h.tasks = sidecar.NewTaskRunner(func(ctx context.Context, r *sdkv1.InvokeRequest) ([]byte, error) {
		close(entered)
		<-release
		out, err := h.executeTask(ctx, r)
		finished <- err
		return out, err
	}, nil, nil)
	b, err := proto.Marshal(&sdkv1.InvokeRequest{FunctionId: "fn.review", Metadata: map[string]string{"taskId": "accepted-before-drain"}})
	require.NoError(t, err)
	_, err = h.handleStartTask(context.Background(), b)
	require.NoError(t, err)
	<-entered
	n, err := h.DrainProviderSession(context.Background(), sess.SessionID, false, time.Second)
	require.NoError(t, err)
	require.Equal(t, 1, n)
	require.EqualValues(t, 0, sess.Inflight())
	close(release)
	select {
	case err := <-finished:
		require.Error(t, err)
		t.Logf("CONFIRMED accepted task failed after drain ACK: %v", err)
	case <-time.After(time.Second):
		t.Fatal("task did not finish")
	}
}
