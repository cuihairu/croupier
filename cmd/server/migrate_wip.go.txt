	defaultConfigFile = "etc/server.yaml"

// migrateCmd represents the migrate command
var migrateCmd = &cobra.Command{
	Use:   "migrate [command]",
	Short: "Database migration tool",
	Long: `数据库迁移命令行工具，支持以下操作：

迁移子命令：
  status        - 查看当前迁移状态
  history       - 查看迁移历史
  up            - 执行向上迁移
  down          - 执行向下迁移
  init          - 初始化迁移表

选项：
  --config      指定配置文件路径 (默认: etc/server.yaml)
  --step        执行指定步骤的迁移
  --force       强制执行迁移，即使已应用
  --dry-run     预览迁移，不实际执行

示例：
  croupier-server migrate status                    # 查看迁移状态
  croupier-server migrate up                       # 执行所有待执行的迁移
  croupier-server migrate up --step 1              # 只执行步骤 1
  croupier-server migrate down --target v1.0.0          # 迁移到指定版本
  croupier-server migrate --dry-run                   # 预览迁移，不实际执行
`,
	Run: func(cmd *cobra.Command, args []string) {
		return runMigrateCommand(cmd, args)
	},
}

// migrateCmdConfig represents configuration for migrate command
type migrateCmdConfig struct {
	configFile string
	force      bool
	dryRun    bool
	step       int
	target     string
	command    string
}

// runMigrateCommand executes the migrate command
func runMigrateCommand(cmd *cobra.Command, args []string) error {
	// 解析命令行参数
	config := parseMigrateFlags(cmd, args)

	// 加载配置
	c, err := config.LoadConfig(config.configFile)
	if err != nil {
		return fmt.Errorf("failed to load config: %w", err)
	}

	// 创建服务上下文
	svcCtx := svc.NewServiceContext(c)

	// 初始化迁移管理器
	migrateManager := migrate.NewMigrationManager(svcCtx.DB)

	// 初始化迁移表
	if err := migrateManager.Init(context.Background()); err != nil {
		return fmt.Errorf("failed to initialize migration table: %w", err)
	}

	switch config.command {
	case "status":
		return showMigrationStatus(migrateManager)
	case "history":
		return showMigrationHistory(migrateManager, config.page, config.pageSize)
	case "up":
		return runMigrationUp(migrateManager, config)
	case "down":
		return runMigrationDown(migrateManager, config)
	case "init":
		return initMigration(migrateManager)
	default:
		return fmt.Errorf("unknown command: %s", config.command)
	}
}

// parseMigrateFlags parses command line flags
func parseMigrateFlags(cmd *cobra.Command, args []string) *migrateCmdConfig {
	config := &migrateCmdConfig{
		configFile: defaultConfigFile,
		command:    "",
	}

	// 设置 flags
	cmd.Flags().BoolVarP(&config.force, "force", "f", false, "强制执行迁移")
	cmd.Flags().BoolVarP(&config.dryRun, "dry-run", "n", false, "预览迁移，不实际执行")
	cmd.Flags().IntVarP(&config.step, "step", "s", -1, "执行指定步骤的迁移")
	cmd.Flags().StringVarP(&config.target, "target", "t", "", "向下迁移到指定版本")
	cmd.Flags().IntVarP(&config.page, "page", "p", 1, "查看历史记录的页码")
	cmd.Flags().IntVarP(&config.pageSize, "page-size", "P", 20, "每页显示的记录数")

	// 解析子命令
	if len(args) > 0 {
		config.command = args[0]
	}

	// 解析配置文件参数
	if file, err := cmd.Flags().GetString("config"); err == nil {
		config.configFile = file
	}

	return config
}

// showMigrationStatus shows current migration status
func showMigrationStatus(migrateManager *migrate.MigrationManager) error {
	fmt.Println("=== 迁移状态 ===")

	latestVersion, err := migrateManager.GetLatestVersion(context.Background())
	if err != nil {
		return fmt.Errorf("failed to get latest version: %w", err)
	}

	pendingCount, err := getPendingMigrationsCount(migrateManager, latestVersion)
	if err != nil {
		return fmt.Errorf("failed to get pending migrations: %w", err)
	}

	fmt.Printf("当前版本: %s\n", latestVersion)
	fmt.Printf("待执行迁移: %d 个\n", pendingCount)

	if pendingCount == 0 {
		fmt.Println("✅ 所有迁移已应用")
	} else {
		fmt.Printf("⏳ 有 %d 个迁移待执行\n", pendingCount)
		fmt.Println("使用 'croupier migrate up' 执行迁移")
	}

	return nil
}

// showMigrationHistory shows migration history
func showMigrationHistory(migrateManager *migrate.MigrationManager, page, pageSize int) error {
	fmt.Println("=== 迁移历史 ===")

	history, err := migrateManager.GetMigrationHistory(context.Background(), pageSize)
	if err != nil {
		return fmt.Errorf("failed to get migration history: %w", err)
	}

	if len(history.Items) == 0 {
		fmt.Println("暂无迁移记录")
		return nil
	}

	fmt.Printf("总记录数: %d\n", history.Total)
	fmt.Printf("当前页: %d, 每页: %d\n", history.Page, history.Size)

	fmt.Println("最近的迁移记录:")
	for i, item := range history.Items {
		fmt.Printf("%d. %s - %s (%s) - %s\n",
			i+1, item.CreatedAt.Format("2006-01-02 15:04:05"),
			item.MigrationName, item.Direction, item.Status,
			item.Duration)
		if item.Error != "" {
			fmt.Printf("    错误: %s\n", item.Error)
		}
	}

	return nil
}

// runMigrationUp runs pending migrations
func runMigrationUp(migrateManager *migrate.MigrationManager, config *migrateCmdConfig) error {
	fmt.Printf("=== 执行向上迁移 ===\n")

	if config.dryRun {
		fmt.Println("🔍 预览模式 - 不会实际执行迁移")
		return nil
	}

	migrations, err := getMigrationsForStep(migrateManager, config.step)
	if err != nil {
		return fmt.Errorf("failed to get migrations: %w", err)
	}

	if len(migrations) == 0 {
		fmt.Println("❌ 没有找到指定的迁移步骤")
		return nil
	}

	fmt.Printf("准备执行 %d 个迁移...\n", len(migrations))

	results := make([]*migrate.MigrationResult, 0, len(migrations))

	for i, migration := range migrations {
		fmt.Printf("执行迁移: %s\n", migration.Name())

		result, err := migrateManager.ExecuteMigration(context.Background(), migration, migrate.DirectionUp)
		results[i] = result

		if err != nil {
			fmt.Printf("❌ 迁移失败: %s - %v\n", migration.Name(), err)
		} else if result.Status == migrate.StatusSuccess {
			fmt.Printf("✅ 迁移成功: %s\n", migration.Name())
		} else {
			fmt.Printf("⚠️ 迁移完成但有警告: %s - %s\n", migration.Name(), result.Status)
		}
	}

	// 显示执行结果
	fmt.Println("\n=== 迁移结果 ===")
	successCount := 0
	failureCount := 0

	for _, result := range results {
		if result.Status == migrate.StatusSuccess {
			successCount++
		} else {
			failureCount++
		}
	}

	fmt.Printf("成功: %d, 失败: %d\n", successCount, failureCount)

	return nil
}
